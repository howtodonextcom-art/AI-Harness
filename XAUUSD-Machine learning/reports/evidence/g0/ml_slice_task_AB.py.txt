def sma(x, k):
    cs = np.cumsum(np.r_[0., x]); out = np.full(len(x), np.nan); out[k - 1:] = (cs[k:] - cs[:-k]) / k; return out

def rolling(fn, x, k):
    out = np.full(len(x), np.nan)
    for i in range(k - 1, len(x)): out[i] = fn(x[i - k + 1:i + 1])
    return out
rng24 = np.full(n, np.nan); rng5 = np.full(n, np.nan)
for i in range(23, n): rng24[i] = h[i - 23:i + 1].max() - l[i - 23:i + 1].min()

sma20 = sma(c, 20); sd20 = np.full(n, np.nan)

feat = {

def wf_predict(model_fns, X, y, ok, years, horizon):
    preds = {k: np.full(n, np.nan) for k in model_fns}
    for Y in years:
        te = np.flatnonzero(ok & (yr == Y))
        if te.size == 0: continue
        first = te[0]
        tr_idx = np.flatnonzero(ok & (np.arange(n) <= first - horizon - 1))
        if tr_idx.size < 3000: continue
        for k, fn in model_fns.items():
            m = fn(); m.fit(X[tr_idx], y[tr_idx]); preds[k][te] = m.predict(X[te])
    return preds

class Mean:
    def fit(self, X, y): self.m = y.mean(); return self
    def predict(self, X): return np.full(len(X), self.m)
class OneFeat:
    def __init__(self, j): self.j = j
    def fit(self, X, y):
        A = np.c_[np.ones(len(X)), X[:, self.j]]; self.b = np.linalg.lstsq(A, y, rcond=None)[0]; return self
    def predict(self, X): return self.b[0] + self.b[1] * X[:, self.j]
class Har:
    def fit(self, X, y): self.m = make_pipeline(StandardScaler(), Ridge(alpha=10.0)).fit(X, y); return self
    def predict(self, X): return self.m.predict(X)
class Lgbm:
    def fit(self, X, y):
        self.m = lgb.LGBMRegressor(n_estimators=200, num_leaves=8, learning_rate=0.05, subsample=0.8, subsample_freq=1,
                                   colsample_bytree=0.8, min_child_samples=50, random_state=7, n_jobs=1, verbose=-1).fit(X, y); return self
    def predict(self, X): return self.m.predict(X)

j14 = FN.index("lATR14")

def mse(p, y, m): return float(np.mean((p[m] - y[m]) ** 2))
def block_ci(pm, pb, y, m, B=2000, seed=7):
    """95% CI of MSE(base) - MSE(model) (positive = model better), monthly block bootstrap."""
    idx = np.flatnonzero(m); mm = mo[idx]; months = np.unique(mm)
    d = ((pb[idx] - y[idx]) ** 2 - (pm[idx] - y[idx]) ** 2)
    bymonth = {u: d[mm == u] for u in months}
    rng = np.random.default_rng(seed); vals = []
    for _ in range(B):
        pick = rng.choice(months, size=len(months), replace=True)
        vals.append(np.concatenate([bymonth[u] for u in pick]).mean())
    return float(np.mean(d)), float(np.percentile(vals, 2.5)), float(np.percentile(vals, 97.5))

                   "ci_LGBM_vs_ATR14": block_ci(predA["M_LightGBM"], predA["B1_ATR14_1feat"], yA, m),

# -------------------------------------------------------------------- Task B (setup quality)
def ema(x, s):
    a = 2 / (s + 1); r = np.empty_like(x); r[0] = x[0]
    for i in range(1, len(x)): r[i] = r[i - 1] + a * (x[i] - r[i - 1])
    return r
e50, e200 = ema(c, 50), ema(c, 200)
slope = e50 - np.r_[np.full(5, np.nan), e50[:-5]]
body = np.abs(c - o); rg = h - l
pin_b = (np.minimum(o, c) - l >= 2 * body) & (h - np.maximum(o, c) <= 0.3 * rg) & (rg > 0.5 * atr14)
pin_s = (h - np.maximum(o, c) >= 2 * body) & (np.minimum(o, c) - l <= 0.3 * rg) & (rg > 0.5 * atr14)
def sh(x, k): r = np.full(n, np.nan); r[k:] = x[:-k]; return r
eng_b = (sh(c, 1) < sh(o, 1)) & (c > o) & (o <= sh(c, 1)) & (c >= sh(o, 1))
eng_s = (sh(c, 1) > sh(o, 1)) & (c < o) & (o >= sh(c, 1)) & (c <= sh(o, 1))
touch_b = (l <= e50 + 0.3 * atr14) & (c > e50); touch_s = (h >= e50 - 0.3 * atr14) & (c < e50)
up = (np.arange(n) > 400) & (c > e50) & (slope > 0); dn = (np.arange(n) > 400) & (c < e50) & (slope < 0)
sig_l = up & touch_b & (pin_b | eng_b); sig_s = dn & touch_s & (pin_s | eng_s)
ev = []

        if r <= 0.3 * atr14[i] or r > 5 * atr14[i]: continue
        tp = e + side * 2 * r
        hs = (l[i + 1:i + 2001] <= sl) if side > 0 else (h[i + 1:i + 2001] >= sl)
        ts_ = (h[i + 1:i + 2001] >= tp) if side > 0 else (l[i + 1:i + 2001] <= tp)
        si = int(np.argmax(hs)) if hs.any() else 10 ** 9; ti = int(np.argmax(ts_)) if ts_.any() else 10 ** 9
        if si == 10 ** 9 and ti == 10 ** 9: continue
        end_bars = min(si, ti)
        ev.append((i, side, 1 if ti < si else 0, end_bars))
ev = np.array(ev)
idxB = ev[:, 0].astype(int); side = ev[:, 1]; yB = ev[:, 2]; dur = ev[:, 3]
er = np.full(n, np.nan); cc = np.abs(c[20:] - c[:-20]); pth = np.convolve(np.abs(np.diff(c)), np.ones(20), "valid"); er[20:] = cc / np.maximum(pth[:len(cc)], 1e-9)

pos50 = (c - lo50) / np.maximum(hi50 - lo50, 1e-9)

def wil(x, p):
    r = np.full(len(x), np.nan); r[p] = x[1:p + 1].mean()
    for i in range(p + 1, len(x)): r[i] = (r[i - 1] * (p - 1) + x[i]) / p
    return r
rsi = 100 - 100 / (1 + wil(np.maximum(rsi_d, 0), 14) / np.maximum(wil(np.maximum(-rsi_d, 0), 14), 1e-9))
FB = np.column_stack([
    side, side * (c[idxB] - e50[idxB]) / atr14[idxB], side * slope[idxB] / atr14[idxB], side * (c[idxB] - e200[idxB]) / atr14[idxB],
    er[idxB], atrp[idxB], feat["lATR14"][idxB], side * (pos50[idxB] - 0.5), rsi[idxB], feat["hs"][idxB], feat["hc"][idxB], feat["ds"][idxB], feat["dc"][idxB],
    side * ((c[idxB] - o[idxB]) / atr14[idxB]),
])
okB = np.isfinite(FB).all(1)
yrB = yr[idxB]
resB = {}
class LogR:
    def fit(self, X, y): self.m = make_pipeline(StandardScaler(), LogisticRegression(C=0.1, max_iter=500)).fit(X, y); return self
    def predict_proba1(self, X): return self.m.predict_proba(X)[:, 1]
class LgbC:
    def fit(self, X, y):
        self.m = lgb.LGBMClassifier(n_estimators=150, num_leaves=6, learning_rate=0.05, subsample=0.8, subsample_freq=1, colsample_bytree=0.8,
                                    min_child_samples=60, random_state=7, n_jobs=1, verbose=-1).fit(X, y); return self
    def predict_proba1(self, X): return self.m.predict_proba(X)[:, 1]
pB = {"base": np.full(len(yB), np.nan), "logit": np.full(len(yB), np.nan), "lgbm": np.full(len(yB), np.nan)}
