yr=df["timestamp"].dt.year().to_numpy(); n=len(c)
pc=np.r_[c[0],c[:-1]]; tr=np.maximum(h-l,np.maximum(abs(h-pc),abs(l-pc)))
cs=np.cumsum(np.r_[0.,tr]); atr=np.full(n,np.nan); atr[13:]=(cs[14:]-cs[:-14])/14
def ema(x,s):
    a=2/(s+1); o_=np.empty_like(x); o_[0]=x[0]
    for i in range(1,len(x)): o_[i]=o_[i-1]+a*(x[i]-o_[i-1])
    return o_
e50=ema(c,50); e200=ema(c,200)
def sh(x,k): r=np.full(n,np.nan); r[k:]=x[:-k]; return r
slope=(e50-sh(e50,5))/atr
warm=np.arange(n)>400
up=warm&(c>e50)&(slope>0); dn=warm&(c<e50)&(slope<0)

pin_s=(h-np.maximum(o,c)>=2*body)&(np.minimum(o,c)-l<=0.3*rng)&(rng>0.5*atr)
eng_b=(sh(c,1)<sh(o,1))&(c>o)&(o<=sh(c,1))&(c>=sh(o,1))
eng_s=(sh(c,1)>sh(o,1))&(c<o)&(o>=sh(c,1))&(c<=sh(o,1))
touch_b=(l<=e50+0.3*atr)&(c>e50); touch_s=(h>=e50-0.3*atr)&(c<e50)
def fw(k):

def tt(a,b,k):
    if len(a)<5: return float('nan')
    return (a.mean()-b.mean())/math.sqrt(a.var(ddof=1)/len(a)+b.var(ddof=1)/len(b))/math.sqrt(k)
def rolling_max(x,k):
    out=np.full(n,np.nan)
    for i in range(k,n): out[i]=x[i-k:i].max()
    return out
def rolling_min(x,k):
    out=np.full(n,np.nan)
    for i in range(k,n): out[i]=x[i-k:i].min()
    return out
def anyk(b,k):
    out=np.zeros(n,bool)
    for j in range(k): out[j:]|=b[:n-j] if j>0 else b
    return out
pre_hi=rolling_max(h,3); pre_lo=rolling_min(l,3)

def trade(m,d,slmode):
    out=[]
    for i in np.flatnonzero(m&np.isfinite(atr)):
        if i+1>=n-1: continue
        e=c[i]
        sl=(l[i-4:i+1].min()-0.1*atr[i]) if d>0 else (h[i-4:i+1].max()+0.1*atr[i]); r=abs(e-sl)
        if r<=0.3*atr[i] or r>5*atr[i]: continue
        tp=e+d*2*r
        hs=(l[i+1:i+1+2000]<=sl) if d>0 else (h[i+1:i+1+2000]>=sl)
        ts=(h[i+1:i+1+2000]>=tp) if d>0 else (l[i+1:i+1+2000]<=tp)
        si=int(np.argmax(hs)) if hs.any() else 10**9; ti=int(np.argmax(ts)) if ts.any() else 10**9
        if si==10**9 and ti==10**9: continue
        out.append((1 if ti<si else 0,int(yr[i]),r/atr[i]))
    return out
def rep(name,ml,ms):
    T=trade(ml,+1,0)+trade(ms,-1,0)
    w=np.array([t[0] for t in T]); y=np.array([t[1] for t in T]); ra=np.array([t[2] for t in T])
    for tag,ys in (("all",np.ones(len(y),bool)),("dev",y<2019),("val",y>=2019)):
        N=int(ys.sum()); p=w[ys].mean(); z=(p-1/3)/math.sqrt(2/9/N)
        print(f"{name:42s} {tag:3s} N={N:5d} win={p:.3f} z={z:+.2f} E[R]={3*p-1:+.3f} medRisk={np.median(ra[ys]):.2f}ATR")
def stats(ml,ms,label):
    for d,m,nm in ((+1,ml,"MUA"),(-1,ms,"BAN")):
        T=trade(m,d,0)
        w=np.array([t[0] for t in T]); y=np.array([t[1] for t in T])
        # thin to non-overlapping: keep at most one entry per 24 bars (approx independence)
        print(f"{label:34s} {nm} N={len(w):5d} win={w.mean():.3f}", " dev/val win=",round(w[y<2019].mean(),3),round(w[y>=2019].mean(),3))

# block bootstrap-ish: effective independence -> keep 1 trade per 24 bars
def thin(m):
    idx=np.flatnonzero(m); keep=[]; last=-999
    for i in idx:
        if i-last>=24: keep.append(i); last=i
    o_=np.zeros(n,bool); o_[keep]=True; return o_
for nm,ml,ms in (("NEN",up,dn),("A",A_l,A_s),("C",C_l,C_s)):
    T=trade(thin(ml),+1,0)+trade(thin(ms),-1,0)
    w=np.array([t[0] for t in T]); N=len(w); p=w.mean(); z=(p-1/3)/math.sqrt(2/9/N)
    print(f"THINNED(1/24 nen) {nm:4s} N={N:5d} win={p:.3f} z={z:+.2f} E[R]={3*p-1:+.3f}")
