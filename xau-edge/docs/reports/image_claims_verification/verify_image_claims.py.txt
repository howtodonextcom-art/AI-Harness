def ema(x, span):
    a = 2.0 / (span + 1.0)
    out = np.empty_like(x)
    out[0] = x[0]
    for i in range(1, len(x)):
        out[i] = out[i - 1] + a * (x[i] - out[i - 1])
    return out

# T1. Fibonacci retracement: P(retest swing before breaking swing) vs 1-r
# ----------------------------------------------------------------------------
def pivots(hh, ll, k):
    m = len(hh)
    wh, wl = swv(hh, 2 * k + 1), swv(ll, 2 * k + 1)
    ph = np.zeros(m, bool)
    pl_ = np.zeros(m, bool)
    ph[k:m - k] = (wh[:, k] >= wh.max(1)) & (wh[:, k] > wh[:, :k].max(1))
    pl_[k:m - k] = (wl[:, k] <= wl.min(1)) & (wl[:, k] < wl[:, :k].min(1))
    both = ph & pl_
    ph &= ~both
    pl_ &= ~both
    seq = [(i, "H", hh[i]) for i in np.flatnonzero(ph)] + [(i, "L", ll[i]) for i in np.flatnonzero(pl_)]
    seq.sort()
    merged = []
    for it in seq:
        if merged and merged[-1][1] == it[1]:
            p = merged[-1]
            if (it[1] == "H" and it[2] > p[2]) or (it[1] == "L" and it[2] < p[2]):
                merged[-1] = it
        else:
            merged.append(it)
    return merged


LEVELS = (0.382, 0.5, 0.618, 0.786, 0.886)


def fib_events(hh, ll, cl, atrv, yr, k=5, min_atr=2.0, wait=500, maxfwd=3000):
    """Entry = first touch AFTER the pivot is confirmed (no look-ahead). Levels already
    breached inside the k-bar confirmation window are not tradable. Entry price = level,
    or the open if the bar gaps through the level."""
    rows = []
    seq = pivots(hh, ll, k)
    for (ia, ta, pa), (ib, tb, pb) in zip(seq[:-1], seq[1:]):
        if not (ta == "L" and tb == "H"):
            continue
        A, B = pa, pb
        H = B - A
        if not np.isfinite(atrv[ib]) or H < min_atr * atrv[ib]:
            continue
        cidx = ib + k
        if cidx + 2 >= len(hh):
            continue
        wmin = ll[ib + 1:cidx + 1].min()
        todo = {r: B - r * H for r in LEVELS if wmin > B - r * H}
        if wmin <= A:
            continue
        j = cidx + 1
        end = min(len(hh), cidx + 1 + wait)
        while j < end and todo:
            if hh[j] > B:
                break
            for r in list(todo):
                lvl = todo[r]
                if ll[j] <= lvl:
                    todo.pop(r)
                    entry = cl[j]
                    if entry - A <= 1e-9 or B - entry <= 1e-9 or ll[j] <= A:
                        continue
                    if True:
                        ls = ll[j + 1:j + 1 + maxfwd] <= A
                        hs = hh[j + 1:j + 1 + maxfwd] >= B
                        si = int(np.argmax(ls)) if ls.any() else 10 ** 9
                        ti = int(np.argmax(hs)) if hs.any() else 10 ** 9
                        if si == 10 ** 9 and ti == 10 ** 9:
                            res = -1
                        elif si <= ti:
                            res = 0
                        else:
                            res = 1
                    if res >= 0:
                        rows.append((r, res, int(yr[j]), (entry - A) / H, 1 if cl[j] >= lvl else 0))
            if ll[j] < A:
                break
            j += 1
    return rows

def sh(x, k):
    out = np.full_like(x, np.nan, dtype=float)
    if k > 0:
        out[k:] = x[:-k]
    return out


def shb(x, k):
    out = np.zeros(len(x), bool)

    return out


def ctx(shift, direction):
    """prior 5-bar move in ATR ending `shift` bars before bar i; direction=-1 => downtrend"""
    mv = (sh(c, shift) - sh(c, shift + 5)) / atr
    return (mv <= -0.5) if direction < 0 else (mv >= 0.5)

def welch(a, b):
    if len(a) < 5 or len(b) < 5:
        return float("nan")
    va, vb = a.var(ddof=1) / len(a), b.var(ddof=1) / len(b)
    return (a.mean() - b.mean()) / math.sqrt(va + vb)

def double_top(hh, ll, cc, atrv, yr, k=5, tol=0.5, gmin=8, gmax=60, depth=1.5, wait=40, maxfwd=3000):
    seq = pivots(hh, ll, k)
    out = []
    for q in range(len(seq) - 2):
        (i1, t1, p1), (im, tm, pm), (i2, t2, p2) = seq[q], seq[q + 1], seq[q + 2]
        if not (t1 == "H" and tm == "L" and t2 == "H"):
            continue
        a = atrv[i2]
        if not np.isfinite(a) or abs(p1 - p2) > tol * a or not (gmin <= i2 - i1 <= gmax):
            continue
        top = max(p1, p2)
        neck = pm
        if top - neck < depth * a:
            continue
        cidx = i2 + k
        if cc[i2:cidx + 1].min() < neck:
            continue
        j = cidx + 1
        end = min(len(cc) - 1, cidx + wait)
        while j <= end:
            if hh[j] > top:
                break
            if cc[j] < neck:
                entry = cc[j]
                sl = top
                tp = neck - (top - neck)
                risk, rew = sl - entry, entry - tp
                if risk <= 0 or rew <= 0:
                    break
                hs = hh[j + 1:j + 1 + maxfwd] >= sl
                ls = ll[j + 1:j + 1 + maxfwd] <= tp
                si = int(np.argmax(hs)) if hs.any() else 10 ** 9
                ti = int(np.argmax(ls)) if ls.any() else 10 ** 9
                if si == 10 ** 9 and ti == 10 ** 9:
                    break
                win = 1 if ti < si else 0
                out.append((win, risk, rew, int(yr[j])))
                break
            j += 1
    return out
