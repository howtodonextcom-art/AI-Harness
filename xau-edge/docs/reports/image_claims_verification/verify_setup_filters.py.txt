def ema(x,s):
    a=2/(s+1); o_=np.empty_like(x); o_[0]=x[0]
    for i in range(1,len(x)): o_[i]=o_[i-1]+a*(x[i]-o_[i-1])
    return o_

def tt(a,b,k):

    return (a.mean()-b.mean())/math.sqrt(a.var(ddof=1)/len(a)+b.var(ddof=1)/len(b))/math.sqrt(k)

        tp=e+d*2*r
        hs=(l[i+1:i+1+2000]<=sl) if d>0 else (h[i+1:i+1+2000]>=sl)
        ts=(h[i+1:i+1+2000]>=tp) if d>0 else (l[i+1:i+1+2000]<=tp)
        si=int(np.argmax(hs)) if hs.any() else 10**9; ti=int(np.argmax(ts)) if ts.any() else 10**9
        if si==10**9 and ti==10**9: continue
        out.append((i,1 if ti<si else 0,int(yr[i])))
    return out
